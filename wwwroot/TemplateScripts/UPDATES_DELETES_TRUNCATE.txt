DECLARE @CANT_AFECTADAS int

BEGIN TRAN 
BEGIN TRY
	--	update
	--	delete 
	--	truncate
	SELECT @CANT_AFECTADAS = @@ROWCOUNT

		IF @CANT_AFECTADAS =1  ---ESTABLECER LA CONDICION PARA HACER EL COMMIT
			BEGIN
				COMMIT TRAN
				SELECT 'HIZO COMMIT - LINEAS AFECTADAS:' + CONVERT(CHAR(10),@CANT_AFECTADAS)
			END
		ELSE 
			BEGIN
				ROLLBACK 
				SELECT  'HIZO ROLLBACK - LINEAS AFECTADAS:'  + CONVERT(CHAR(10),@CANT_AFECTADAS)
			END
END TRY 
BEGIN CATCH
		SELECT  'HIZO ROLLBACK'  
                    SELECT ERROR_MESSAGE() 
		ROLLBACK TRAN
END CATCH
